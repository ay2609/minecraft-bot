```typescript
eventBus.on('tactical:queue-ready', (queue) => {
  // Dequeue first action and execute
  const nextAction = queue.actions[0];
  if (nextAction) {
    void executor.executeAction(nextAction).catch((err: unknown) => {
      console.error('[Executor] Unexpected error from executeAction:', err);

  }
});
```
